ingredients:

    name_en: Pork tenderloin, sliced
    amount: 150g
  - name_th: กะทิไขมันต่ำ
    name_en: Light coconut milk
    amount: 150ml
  - name_th: พริกแกงแดง
    name_en: Red curry paste
    amount: 2 tbsp
  - name_th: พริกชี้ฟ้าแดง
    name_en: Red spur chili
    amount: 2, sliced
  - name_th: ใบมะกรูด
    name_en: Kaffir lime leaves
    amount: 3 leaves
  - name_th: ใบโหระพา
    name_en: Thai basil leaves
    amount: 1/2 cup
steps:
  - th: ตั้งกะทิให้แตกมัน ใส่พริกแกงผัดจนหอม
    en: Simmer coconut milk until the oil separates, add curry paste and fry until

  - th: ใส่หมูสันใน ผัดจนสุกทั่ว
    en: Add pork tenderloin, stir until fully cooked
  - th: เติมกะทิที่เหลือและใบมะกรูด ต้มพอเดือด
    en: Add remaining coconut milk and kaffir lime leaves, bring to a light boil
  - th: ใส่พริกชี้ฟ้าและใบโหระพา ปิดไฟ เสิร์ฟ
    en: Add chili and basil, turn off heat, serve
